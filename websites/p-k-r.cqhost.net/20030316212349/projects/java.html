public static Hashtable parse(String a)

//this is for parsing strings through the program using the token method


{ String field[]=new String[20];
String fieldin[]=new String[20];
Hashtable hash = new Hashtable();
StringTokenizer tokenizer =new StringTokenizer(a,"&");

int i=0;

while (tokenizer.hasMoreTokens())
{
field[i] = tokenizer.nextToken();
StringTokenizer tokenizerin =new StringTokenizer(field[i],"=");
int k=0;
fieldin[0] = null;
fieldin[1] = null;
while (tokenizerin.hasMoreTokens())
{
fieldin[k] = java.net.URLDecoder.decode(tokenizerin.nextToken());
k++;
}
if (fieldin[1]!=null) hash.put(fieldin[0], fieldin[1]);
i++;
}

return hash;

}

public static void returnallfunc(String returnall)
{ Statement statement;
ResultSet resultSet;

ResultSetMetaData metaData;

int columnCount;
int i;

String query = "SELECT * FROM "+returnall;
try {
statement = connection.createStatement();
resultSet = statement.executeQuery(query);
metaData = resultSet.getMetaData();
columnCount = metaData.getColumnCount();
System.out.print("<html><head><title>Show All Record</title>" +

"</head><body><table border=\"1\" height=\"76\">");
System.out.print("<tr>");
for (i=1; i<=columnCount; i++)
{

System.out.print("<td>"+metaData.getColumnName(i)+"</td>");

}
System.out.print("</tr>");

while (resultSet.next())
{System.out.print("<tr>");
for (i=1; i<=columnCount; i++)
{
if (metaData.getColumnName(i).equals("START_DATE") || metaData.getColumnName(i).equals("RETURN_DATE"))
System.out.print("<td>"+resultSet.getString(i).substring(0,10)+"</td>");
else System.out.print("<td>"+resultSet.getString(i)+"</td>");



}
System.out.println("</tr>");
}
}
